从定义到交付

一条工作流，贯穿配置全程

数据只定义一次，规则只写一次，问题只定位一次。七个阶段顺流而下。

  1. 01

    DEFINE

    用 CFT 定义类型

    先说清每种数据长什么样：字段、类型、默认值、引用。CFT 能表达的不止字段——枚举、继承、集合都写得清楚。点标签或后面的卡片切换。

    schema/skills.cftCFT
    @label("技能")
type Skill {
  name: string;
  element: Element;
  power: int;
  cooldown: float = 0.0;
  is_passive: bool = false;
  follow_up: &Skill? = null;
}
    schema/enums.cftCFT
    @flag
@label("权限")
enum Permission {
  @label("读取")  Read = 1,
  @label("写入")  Write = 2,
  @label("执行")  Execute = 4,
}
    schema/rewards.cftCFT
    abstract type Reward {
  check { id != ""; }
}
type ItemReward : Reward {
  item: Item;
  count: int = 1;
}
sealed type CurrencyReward : Reward {
  amount: int;
}
    schema/loot.cftCFT
    type LootTable {
  item_ids: [string] = [];
  weights: [int] = [];
  resistances: {DamageType: float} = {};
  bonus: &Item? = null;
}
  2. 02

    CONSTRAIN

    用 check 写下规则

    把业务规则直接写在数据旁边，还能附上校验失败时的提示。基础比较、条件分支、集合量化——编辑和构建都遵守同一套规则。

    schema/skills.cftCFT
    check {
  power >= 0: "技能威力不能为负";
  cooldown >= 0.0;
  tags.isUnique(): "标签不能重复";
  name != "": "技能名不能为空";
}
    schema/skills.cftCFT
    check {
  when !is_passive {
    power > 0: "主动技能威力需大于 0";
    cooldown > 0.0: "主动技能需要冷却时间";
  }
}
    schema/loot.cftCFT
    check {
  all w in weights { w >= 0; }
  any w in weights { w > 0; }
  none id in item_ids { id == ""; }
}
