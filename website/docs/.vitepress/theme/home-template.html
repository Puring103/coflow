  3. 03

    MODEL

    建立统一的数据模型

    用 CFT 描述类型、字段与引用，让策划、AI和程序遵循共同的数据语言。

    skills.cftCFT
    enum Element { Arcane, Fire, Ice }
table Skill {
  @localized name: string;
  element: Element = Element::Arcane;
  tier: int;
  power: int = 0;
  cooldown: float = 0.0;
  prerequisites: [&Skill] = [];
  @dimension("platform") asset_bundle: string = "skills_pc.bundle";
}
    schema/enums.cftCFT
    @label("稀有度")
enum Rarity {
  @label("普通") Common = 0,
  @label("稀有") Rare = 1,
  @label("史诗") Epic = 2,
  @label("传说") Legendary = 3,
}
    schema/skills.cftCFT
    # 记录 ID 生成强类型 SkillId 枚举
enum SkillId {}
@idAsEnum(SkillId)
@label("技能")
@description("战斗技能配置")
table Skill {
  # 按语言维度保存本地化名称
  @localized
  @label("名称")
  name: string = "未命名";
}
    schema/rewards.cftCFT
    table Item { name: string; }
abstract data Reward {
  label: string;
}
data ItemReward : Reward {
  item: &Item;
  count: int = 1;
}
sealed data CurrencyReward : Reward {
  amount: int = 100;
}
  4. 04

    CHECK

    为数据定义规则

    在 CFT 中声明类型内规则和跨记录检查，让人和 AI 的修改都遵循相同的业务约束。

    skills.cft · SkillCFT
    check Progression {
  power >= 0: "技能威力不能为负";
  all required in prerequisites {
    tier > required.tier:
      "技能阶级必须高于前置技能";
  }
}
    skills.cft · SkillCFT
    check AdvancedSkill {
  when tier > 1 {
    prerequisites.len() > 0:
      "进阶技能需要前置技能";
  }
}
    skills.cft · SkillCFT
    check Prerequisites {
  all required in prerequisites {
    tier > required.tier:
      "技能阶级必须高于前置技能";
  }
}
    schema/project-checks.cftCFT
    # 跨记录的项目级完整性规则
check SkillIntegrity {
  records(Skill).len() > 0:
    "项目中至少需要一个技能";

  all skill in records(Skill) {
    skill.power >= 0:
      "技能威力不能为负";
  }
}
