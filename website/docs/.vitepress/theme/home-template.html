  1. 01

    MODEL

    建立统一的数据模型

    使用专门设计的语言描述数据、字段、类型、枚举，让程序、策划和 AI 遵循共同的数据语言。

    schema/skills.cftCFT
    @idAsEnum(SkillId)
@label("技能")
type Skill {
  name: string = "未命名技能";
  element: Element = Element.Fire;
  power: int = 0;
  cooldown: float = 0.0;
  tags: [string] = [];
  resistance: {Element: float} = {};
  is_passive: bool = false;
  follow_up: &Skill? = null;
}
    schema/enums.cftCFT
    @label("稀有度")
enum Rarity {
  @label("普通") Common = 0,
  @label("稀有") Rare = 1,
  @label("史诗") Epic = 2,
  @label("传说") Legendary = 3,
}
    schema/items.cftCFT
    # record key 生成强类型 SkillId 枚举
@idAsEnum(SkillId)
@label("技能")
@description("战斗技能配置")
type Skill {
  # 按语言维度保存本地化名称
  @localized
  @label("名称")
  name: string = "未命名";
}
    schema/rewards.cftCFT
    abstract type Reward {
  id: string = "reward_001";
  check { id != ""; }
}
type ItemReward : Reward {
  item: Item;
  count: int = 1;
}
sealed type CurrencyReward : Reward {
  amount: int = 100;
}
  2. 02

    CHECK

    为数据定义规则

    在 CFT 内直接定义数据需要遵循的规则，保证数据的正确性。

    schema/skills.cftCFT
    check {
  power >= 0: "技能威力不能为负";
  cooldown >= 0.0;
  tags.isUnique(): "标签不能重复";
  name != "": "技能名不能为空";
}
    schema/skills.cftCFT
    check {
  when !is_passive {
    power > 0: "主动技能威力需大于 0";
    cooldown > 0.0: "主动技能需要冷却时间";
  }
}
    schema/loot.cftCFT
    check {
  all w in weights { w >= 0; }
  any w in weights { w > 0; }
  none id in item_ids { id == ""; }
}
    schema/project-checks.cftCFT
    # 跨记录的项目级完整性规则
check SkillIntegrity {
  records(Skill).len() > 0:
    "项目中至少需要一个技能";

  all skill in records(Skill) {
    skill.power >= 0:
      "技能威力不能为负";
  }
}

    DIMENSION

    维度和本地化

    引入了维度的概念，让同一个字段有不同的变体。通过这个特性，我们可以简单地实现任意数据的本地化。

    维度配置配置 · CFT · 具体数据
    coflow.yaml
    dimensions:
  platform:
    variants: [windows, android, ios]
    out_dir: data/dimensions/platform
  language:
    variants: [zh, en]
    out_dir: data/dimensions/language
    schema/skills.cft
    @dimension("platform")
asset_bundle: string;

@localized
name: string;
